```js
// Shared helpers for the check scripts. Node built-ins only.
import { readdirSync, readFileSync, statSync, existsSync } from 'node:fs';
import { join, relative, sep } from 'node:path';

export const ROOT = process.cwd();

/** Recursively list files under dir (relative to ROOT) whose extension is in exts. */
export function walk(dir, exts) {
  const abs = join(ROOT, dir);
  if (!existsSync(abs)) return [];
  const out = [];
  for (const name of readdirSync(abs)) {
    if (name === 'node_modules' || name.startsWith('.')) continue;
    const full = join(abs, name);
    if (statSync(full).isDirectory()) out.push(...walk(relative(ROOT, full), exts));
    else if (exts.some((e) => name.endsWith(e)))
      out.push(relative(ROOT, full).split(sep).join('/'));
  }
  return out.sort();
}

export function read(rel) {
  return readFileSync(join(ROOT, rel), 'utf8');
}

/** Flatten a nested copy object into [ ['a.b.c', 'value'], ... ]. */
export function flatten(obj, prefix = '') {
  const out = [];
  for (const [k, v] of Object.entries(obj)) {
    const key = prefix ? `${prefix}.${k}` : k;
    if (v && typeof v === 'object' && !Array.isArray(v)) out.push(...flatten(v, key));
    else if (Array.isArray(v)) v.forEach((item, i) => out.push([`${key}.${i}`, String(item)]));
    else out.push([key, String(v)]);
  }
  return out;
}
```

```js
import { existsSync } from 'node:fs';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { ROOT, flatten } from './lib.js';

export const STUDENT_BANNED = [
  /\bdiagnos\w*/i,
  /\bdisorders?\b/i,
  /\bdepress\w*/i,
  /\banxiety\b/i,
  /\banxious\b/i,
  /\bsymptoms?\b/i,
  /\bpatients?\b/i,
  /\bclinical\b/i,
  /\bclinicians?\b/i,
  /\btherapy\b/i,
  /\btreatments?\b/i,
  /\bscreening\b/i,
  /\bassessments?\b/i,
  /\brisks?\b/i,
  /\bseverity\b/i,
  /\bscor(e|es|ed|ing)\b/i,
  /\bpriorit(y|ies)\b/i,
  /\btriage\b/i,
  /\bP[123]\b/,
  /\bFor Review\b/i,
  /\blabels?\b/i,
  /\bflag\w*/i,
];
export const STAFF_BANNED = [
  /\bdiagnos\w*/i,
  /\bdisorders?\b/i,
  /\bdepress\w*/i,
  /\banxiety\b/i,
  /\bsymptoms?\b/i,
  /\bpatients?\b/i,
  /\brisk scores?\b/i,
  /\brisk levels?\b/i,
  /\bat-risk\b/i,
  /\bscreening\b/i,
  /\bseverity\b/i,
];

/** Returns a list of { key, value, pattern } violations. */
export function findViolations(copy, allowlist) {
  const out = [];
  for (const [key, value] of flatten(copy)) {
    if (allowlist.includes(value)) continue;
    const staff = key.startsWith('staff.') || key.startsWith('admin.');
    for (const re of staff ? STAFF_BANNED : STUDENT_BANNED) {
      if (re.test(value)) out.push({ key, value, pattern: String(re) });
    }
  }
  return out;
}

if (process.argv[1] && process.argv[1].endsWith('check-copy.js')) {
  const enPath = join(ROOT, 'src/copy/en.js');
  const allowPath = join(ROOT, 'src/copy/allowlist.js');
  if (!existsSync(enPath)) {
    console.log('check:copy: src/copy/en.js not created yet (Phase 02) — nothing to check');
  } else {
    const { en } = await import(pathToFileURL(enPath).href);
    const allowlist = existsSync(allowPath)
      ? (await import(pathToFileURL(allowPath).href)).ALLOWLIST
      : [];
    const v = findViolations(en, allowlist);
    if (v.length) {
      for (const x of v) console.error(`check:copy: ${x.key} matches ${x.pattern}: "${x.value}"`);
      process.exit(1);
    }
    console.log('check:copy: OK');
  }
}
```

**`scripts/check-colour-words.js`**

```js
// 00 Section 10.4. Fails if a traffic-light colour word appears outside the allowed places.
// Inside src/config/vocabulary.js and src/copy/en.js, colour words are allowed ONLY between the
// marker comments  // office-tags:start  and  // office-tags:end .
import { walk, read } from './lib.js';

export const COLOUR = /\b(green|yellow|red|amber|emerald|lime|crimson|scarlet)\b/i;

const ALLOWED_WHOLE_FILE = [
  /^supabase\/migrations\/[^/]*_office_tags[^/]*\.sql$/,
  /^src\/features\/guidance\/OfficeTag[^/]*\.jsx$/,
  /^src\/features\/guidance\/office-tag\.css$/,
];
const MARKER_FILES = ['src/config/vocabulary.js', 'src/copy/en.js'];

/** Returns violations [{file, line, text}] for one file's contents. */
export function scanFile(file, text) {
  if (ALLOWED_WHOLE_FILE.some((re) => re.test(file))) return [];
  const markers = MARKER_FILES.includes(file);
  let inside = false;
  const out = [];
  text.split('\n').forEach((line, i) => {
    if (markers && line.includes('office-tags:start')) inside = true;
    if (!(markers && inside) && COLOUR.test(line))
      out.push({ file, line: i + 1, text: line.trim() });
    if (markers && line.includes('office-tags:end')) inside = false;
  });
  return out;
}

if (process.argv[1] && process.argv[1].endsWith('check-colour-words.js')) {
  const files = [
    ...walk('src', ['.js', '.jsx', '.css', '.html', '.json']),
    ...walk('supabase/functions', ['.js', '.json', '.ts']),
    ...walk('supabase/migrations', ['.sql']),
  ];
  const v = files.flatMap((f) => scanFile(f, read(f)));
  if (v.length) {
    for (const x of v) console.error(`check:colours: ${x.file}:${x.line}: ${x.text}`);
    process.exit(1);
  }
  console.log(`check:colours: OK (${files.length} files scanned)`);
}
```

## 8. Step 6 — Migrations

Create these seven files in `supabase/migrations/` with exactly this content. The file names fix the order. Never edit a migration after it is committed; later phases add new migration files.

Rules the migrations implement (read before copying, so you do not "fix" them):
- **Start from zero.** Migration 500 revokes every table privilege from `anon` and `authenticated`, then grants back only what each policy needs, **by column**. A client can never send a column it was not granted (for example `created_at`, `disclosed`, `assigned_by`, `status` on insert).
- **`support_requests.note` is not granted for SELECT.** The only way to read a note is `open_support_request()`, which writes the `view_request` audit row first. In later phases, never write `select('*')` on `support_requests`; it fails with "permission denied".
- **Students cannot insert support requests at all.** Only `create_support_request()` creates them, and only the `service_role` may call it (from the Phase 07 Edge Function).
- **`disclosed` is always built in SQL** from `students` (and the latest `checkins` row for requests). Clients cannot send it.
- **Audit rows are written by the database** (triggers and RPCs), so a screen cannot forget to log. The UI never inserts into `staff_access_audit` directly.
- **Colour words** (`green`, `yellow`, `red`) appear only in the `office_tags` migration (00 Section 10.4).
- **`aal2` is required** for every guidance and administrator permission, through `is_guidance()` and `is_admin()`.

## 13. Data access contract (later phases must follow this exactly)

This is the only allowed way each role touches each table. "Direct" means supabase-js from the browser under RLS. A phase may not use any other path.

| Table / object | Student | Guidance (aal2) | Administrator (aal2) | Anonymous |
|---|---|---|---|---|
| `students` | read own (direct) | none | read all (direct) | none |
| `consent_records` | read own; insert own **assent** (direct) | none | read all; insert (parental or assent); update `status` to withdrawn (direct) | none |
| `checkins` | read own; insert own if Tier 2 (direct) | none | none | none |
| `prompt_events` | read own; insert own (direct) | none | none | none |
| `resource_views` | read own; insert own if Tier 2 (direct) | none | none | none |
| `resources` | read published (direct) | read published (direct) | read all; insert; update; delete (direct) | read published (direct) |
| `crisis_items` | read active (direct) | read all (direct) | read all; insert; retire via `is_active=false` (direct) | none |
| `crisis_events` | insert own: `{student_id, trigger_item_id}` only, **never chain `.select()`** (direct) | read via `crisis_channel` view; `log_crisis_view`, `acknowledge_crisis_event`, `resolve_crisis_event` RPCs | none | none |
| `support_requests` | none (created only by Phase 07 Edge Function → `create_support_request`) | list via `support_queue` view; open via `open_support_request` RPC; update `status`, `follow_up_at` (direct) | none | none |
| `triage_decisions` | none | read (via `open_support_request` / `support_queue`) | none | none |
| `office_tags` | none | insert `{request_id, tag}`; read (direct) | none | none |
| `staff_access_audit` | none | read (direct); writes happen automatically | read (direct) | none |
| `staff_accounts` | none | read own row (any aal); read all at aal2 | read all; update `role`, `is_active`, `display_name` (direct); create via Phase 09 Edge Function | none |
| `my_account()` RPC | yes | yes | yes | no |

RPC return shapes the UI relies on:

- `my_account()` → student: `{ kind: 'student', full_name, grade_level, section, class_adviser, assent: 'granted'|'withdrawn'|'none', parental: 'granted'|'withdrawn'|'none', tier2: boolean, has_checkin: boolean }`; staff: `{ kind: 'guidance'|'administrator', is_active, display_name }`; signed-in with neither row: `{ kind: 'none' }`.
- `open_support_request(id)` → `{ id, created_at, status, concern_type, follow_up_at, note, disclosed: { full_name, grade_level, section, class_adviser, c1, c2, c3, c4 }, triage: { score, band, guardrails, computed_at } | null, office_tag: { tag, assigned_at, assigned_by } | null, has_other_open_request, viewed_at }`.
- `support_queue` view columns: `id, created_at, status, concern_type, follow_up_at, full_name, grade_level, section, class_adviser, band, has_other_open_request, first_contact_at, is_overdue`.
- `crisis_channel` view columns: `id, created_at, full_name, grade_level, section, class_adviser, protocol_ref, item_label, item_set_version, acknowledged_at, acknowledged_by, resolved_at, resolved_by`.

| Code | Meaning | Where |
|---|---|---|

| `23514` | A rule was broken (CHECK, retired crisis item, resolve before acknowledge) | forms, crisis |

| `40001` | `stale_checkin`: a newer check-in exists | Phase 07 |
